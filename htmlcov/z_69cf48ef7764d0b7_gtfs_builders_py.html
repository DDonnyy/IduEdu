245def _scheduled_wait_times( 

246 stop_times: pd.DataFrame, 

247 frequency_trip_ids: set[str], 

248 start: float | None, 

249 end: float | None, 

250 trips: pd.DataFrame | None = None, 

251) -> tuple[pd.Series, pd.MultiIndex]: 

252 """Half the mean gap between departures a rider at the stop can take, per pattern-stop. 

253 

254 The gap is taken between the departures of one **line leaving the stop for the same 

255 next stop**: every pattern of the route that goes on to that stop, wherever it began 

256 or ends. The patterns of a route split its trips by every difference -- branches 

257 beyond a shared section, short workings, an express, a slightly different shape -- 

258 and a gap within one pattern is the wait as if the others did not run: at Auber the 

259 westbound RER A runs about 20 trains an hour over 6 patterns, and per pattern the 

260 wait came out at 4-12 minutes against about 1.5. The next stop, not ``direction_id``, 

261 decides the direction: some feeds put both directions under one ``direction_id``. 

262 Beyond a branching point the next stops differ and the branches separate again. 

263 Each pattern-stop gets the wait of its line there; without ``trips`` (no route 

264 ids), a pattern is its own line. 

265 """ 

266 names = ["pattern_id", "stop_position"] 

267 boardable = ~stop_times["pickup_type"].eq("1") 

268 if frequency_trip_ids: 

269 boardable &= ~stop_times["trip_id"].isin(frequency_trip_ids) 

270 boardable &= _window_mask(stop_times["departure_seconds"], start, end) 

271 columns = ["trip_id", "pattern_id", "stop_position", "stop_id", "departure_seconds"] 

272 departures = stop_times.loc[boardable, columns].copy() 

273 if departures.empty: 

274 empty_index = pd.MultiIndex.from_arrays([[], []], names=names) 

275 return pd.Series(index=empty_index, dtype=float), empty_index 

276 

277 # the next stop of every pattern-stop, from the pattern's own sequence 

278 sequence = stop_times[["pattern_id", "stop_position", "stop_id"]].drop_duplicates(subset=names) 

279 following = sequence.assign(stop_position=sequence["stop_position"] - 1).rename(columns={"stop_id": "next_stop"}) 

280 departures = departures.merge(following[names + ["next_stop"]], on=names, how="left") 

281 departures["next_stop"] = departures["next_stop"].fillna("") 

282 if trips is not None and "route_id" in trips: 282 ↛ 288line 282 didn't jump to line 288 because the condition on line 282 was always true

283 route_of = trips.drop_duplicates("trip_id").set_index("trip_id")["route_id"].astype(str) 

284 departures["line"] = departures["trip_id"].map(route_of).fillna("") 

285 missing = departures["line"].eq("") 

286 departures.loc[missing, "line"] = "pattern:" + departures.loc[missing, "pattern_id"].astype(str) 

287 else: 

288 departures["line"] = "pattern:" + departures["pattern_id"].astype(str) 

289 

290 line_key = ["line", "stop_id", "next_stop"] 

291 times = departures[line_key + ["departure_seconds"]].drop_duplicates().sort_values(line_key + ["departure_seconds"]) 

292 groups = times.groupby(line_key, sort=False)["departure_seconds"] 

293 counts = groups.size().rename("count") 

294 gaps = times.assign(gap=groups.diff()).dropna(subset=["gap"]) 

295 line_wait = (gaps.groupby(line_key, sort=False)["gap"].mean() / 120).rename("wait") 

296 

297 pattern_stops = departures[names + line_key].drop_duplicates(subset=names) 

298 pattern_stops = pattern_stops.merge(counts, left_on=line_key, right_index=True, how="left") 

299 pattern_stops = pattern_stops.merge(line_wait, left_on=line_key, right_index=True, how="left") 

300 indexed = pattern_stops.set_index(names) 

301 waits = indexed["wait"].dropna().astype(float) 

302 single = indexed.index[indexed["count"].eq(1)] 

303 return waits, single

372def _average_day_waits(

379) -> tuple[pd.Series, pd.MultiIndex]: 

380 """Each service day's waits on its own trips, averaged with every day weighing the same. 

381 

382 A pattern-stop is averaged over the days it runs; one with a single departure on

871def get_gtfs_public_transport_graph( 

872 feed: str | Path | GTFSFeed | Sequence[str | Path | GTFSFeed], 

873 *, 

874 service_date: date | datetime | str | None = None, 

875 service_period: tuple[date | datetime | str, date | datetime | str] | None = None, 

876 service_dates: Sequence[date | datetime | str] | None = None, 

877 start_time: time | str | int | float | None = None, 

878 end_time: time | str | int | float | None = None, 

879 crs: Any | None = None, 

880 single_departure_wait_min: float | None = None, 

881 walk_speed_m_per_min: float = DEFAULT_WALK_SPEED_M_PER_MIN, 

882) -> UrbanGraph: 

883 """Build a static public-transport ``UrbanGraph`` from a local GTFS feed. 

884 

885 Boarding weights use the first-iteration half-headway heuristic. Scheduled 

886 services use half the mean gap between consecutive departures of the same 

887 route pattern at the same stop. ``frequencies.txt`` uses 

888 ``headway_secs / 120``. 

889 

890 A gap is a gap between departures of the same day. With ``service_date`` the 

891 graph is that day's timetable. Without it each pattern-stop gets the **average 

892 day's** wait: the wait of every service day of ``service_period`` -- or of the 

893 feed's own calendar, at most 366 days from its start -- computed 

894 on that day's trips alone, then averaged with every calendar day weighing the same, 

895 over the days the pattern-stop runs. Trips running on none of those days are left 

896 out. A feed without ``calendar``/``calendar_dates`` is taken as a single day. 

897 

898 ``start_time``/``end_time`` bound the departures within each day; ``None`` means 

899 unbounded. 

900 

901 Args: 

902 feed: Directory containing GTFS text files, a GTFS ZIP archive, an 

903 already-read :class:`iduedu.gtfs.GTFSFeed`, or a sequence of any of those. A 

904 sequence is merged with :func:`iduedu.merge_gtfs_feeds` first, which 

905 qualifies identifiers by source; pass the merged feed yourself when 

906 you need to control prefixes or fuse stops across feeds. 

907 service_date: Optional service date used with ``calendar.txt`` and 

908 ``calendar_dates.txt``. Accepts ``date``, ``YYYY-MM-DD`` or 

909 ``YYYYMMDD``. 

910 service_period: Optional ``(first, last)`` dates, both inclusive, to average 

911 the waits over instead of the feed's own calendar (for example a reference 

912 week). Cannot be combined with ``service_date``. 

913 service_dates: Optional explicit days to average the waits over, e.g. a 

914 reference week plus the own weeks of merged sources not in service then. 

915 Every day still holds only its own trips, so days of different years do 

916 not mix. Cannot be combined with ``service_date`` or ``service_period``. 

917 start_time: Inclusive departure-time bound. ``None`` means unbounded. 

918 end_time: Exclusive departure-time bound. ``None`` means unbounded. 

919 crs: Projected target CRS. If omitted, a local UTM CRS is estimated. 

920 single_departure_wait_min: Explicit wait fallback for a pattern-stop 

921 with one scheduled departure. If ``None``, its boarding edge is 

922 omitted. 

923 walk_speed_m_per_min: Fallback speed for pathways without an explicit 

924 traversal time. 

925 """ 

926
